#include <stdio.h>
#include <stdlib.h>
#include <dirent.h>
#include <ctype.h>
#include <unistd.h>

extern void log_action(int pid, const char* action);

int is_numeric(const char* str) {
    for (int i = 0; str[i] != '\0'; i++) {
        if (!isdigit(str[i])) return 0;
    }
    return 1;
}

void check_memory_limit(int pid, const char* comm) {
    char path[256];
    snprintf(path, sizeof(path), "/proc/%d/statm", pid);
    
    FILE *fp = fopen(path, "r");
    if (fp) {
        long size, resident;
        if (fscanf(fp, "%ld %ld", &size, &resident) == 2) {
           long memory_mb = (resident * sysconf(_SC_PAGESIZE)) / (1024 * 1024);
            if (memory_mb > 500) { // Limit hardcoded to 500MB
                log_action(pid, "WARNING: MEMORY LIMIT EXCEEDED");
            }
        }
        fclose(fp);
    }
}

void send_process_list(int fd_out) {
    FILE *data_out = fdopen(fd_out, "w"); // Wrap pipe in FILE* for fprintf
    struct dirent *entry;
    DIR *dp = opendir("/proc");

    fprintf(data_out, "%-10s %-10s %-10s %-20s\n", "PID", "PPID", "STATE", "NAME");
    fprintf(data_out, "------------------------------------------------------\n");

    if (dp != NULL) {
        while ((entry = readdir(dp)) != NULL) {
            if (entry->d_type == DT_DIR && is_numeric(entry->d_name)) {
                char path[256];
                snprintf(path, sizeof(path), "/proc/%s/stat", entry->d_name);
                
                FILE *fp = fopen(path, "r");
                if (fp) {
                    int pid, ppid;
                    char comm[256], state;
                    if (fscanf(fp, "%d (%[^)]) %c %d", &pid, comm, &state, &ppid) == 4) {
                        fprintf(data_out, "%-10d %-10d %-10c %-20s\n", pid, ppid, state, comm);
                        check_memory_limit(pid, comm);
                    }
                    fclose(fp);
                }
            }
        }
        closedir(dp);
    }
    fprintf(data_out, "------------------------------------------------------\n");
    fprintf(data_out, "END\n"); // Tell the UI we are done sending
    fflush(data_out); // Force data through the pipe instantly
}
