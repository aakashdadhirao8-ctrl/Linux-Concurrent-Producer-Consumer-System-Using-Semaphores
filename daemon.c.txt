#include <unistd.h>
#include <stdio.h>


extern void send_process_list(int fd_out);
extern void handle_control(int pid, int action);

void run_monitor_daemon(int fd_in, int fd_out) {
    int cmd[2]; // cmd[0] = action, cmd[1] = pid

 
    while (read(fd_in, cmd, sizeof(cmd)) > 0) {
        
        if (cmd[0] == -1) {
            break; // Exit command
        }
        
        if (cmd[0] == 0) {
        
            send_process_list(fd_out);
        } 
        else {
      
            handle_control(cmd[1], cmd[0]);
        }
    }
}
