#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <string.h>
#include <sys/wait.h>


extern void run_monitor_daemon(int fd_in, int fd_out);

int main() {
    int pipe_cmd[2];  // Pipe for commands: UI -> Daemon
    int pipe_data[2]; // Pipe for data: Daemon -> UI

    if (pipe(pipe_cmd) == -1 || pipe(pipe_data) == -1) {
        perror("Pipe creation failed");
        return 1;
    }

    pid_t pid = fork();

    if (pid < 0) {
        perror("Fork failed");
        return 1;
    }

    if (pid == 0) {
     
        close(pipe_cmd[1]);  
        close(pipe_data[0]); 
        
        run_monitor_daemon(pipe_cmd[0], pipe_data[1]);
        
        close(pipe_cmd[0]);
        close(pipe_data[1]);
        exit(0);
    } 
    else {
        close(pipe_cmd[0]);  
        close(pipe_data[1]); 

        FILE *data_in = fdopen(pipe_data[0], "r");
        char buffer[256];
        int cmd[2]; // cmd[0] = action, cmd[1] = pid

        while(1) {
            system("clear");
            printf("=== LINUX MULTI-FILE IPC MONITOR ===\n\n");

        
            cmd[0] = 0; cmd[1] = 0;
            write(pipe_cmd[1], cmd, sizeof(cmd));

            while (fgets(buffer, sizeof(buffer), data_in) != NULL) {
                if (strcmp(buffer, "END\n") == 0) break;
                printf("%s", buffer);
            }

       
            printf("\nOptions: [0] Refresh | [PID] Control Process | [-1] Exit\nChoice: ");
            
            int input_pid;
            if (scanf("%d", &input_pid) != 1) {
                while(getchar() != '\n'); 
                continue;
            }

            if (input_pid == -1) {
                cmd[0] = -1; 
                write(pipe_cmd[1], cmd, sizeof(cmd)); 
                break;
            } else if (input_pid > 0) {
                printf("Action for PID %d -> 1:STOP 2:CONT 3:KILL : ", input_pid);
                if (scanf("%d", &cmd[0]) == 1) {
                    cmd[1] = input_pid;
                    write(pipe_cmd[1], cmd, sizeof(cmd)); // Send kill/stop cmd
                    
                    printf("Signal command sent via IPC pipe...\nPress Enter to continue...");
                    while(getchar() != '\n'); getchar(); 
                }
            }
        }
        
     
        fclose(data_in);
        close(pipe_cmd[1]);
        printf("System shut down cleanly.\n");
    }
    return 0;
}
