#include <stdio.h>
#include <signal.h>
#include <time.h>
#include <string.h>

void log_action(int pid, const char* action) {
    FILE *log_file = fopen("os_project.log", "a");
    if (log_file) {
        time_t now = time(NULL);
        char *time_str = ctime(&now);
        time_str[strlen(time_str)-1] = '\0';
        fprintf(log_file, "[%s] PID: %d | Action: %s\n", time_str, pid, action);
        fclose(log_file);
    }
}

void handle_control(int pid, int action) {
    int sig = 0;
    const char* act_str = "";
    
    if (action == 1) { sig = SIGSTOP; act_str = "STOPPED"; }
    else if (action == 2) { sig = SIGCONT; act_str = "CONTINUED"; }
    else if (action == 3) { sig = SIGKILL; act_str = "KILLED"; }

    if (sig != 0) {
        if (kill(pid, sig) == 0) {
            log_action(pid, act_str);
        } else {
            // Note: Since this is a background process, printing perror 
            // will mess up the UI temporarily until a refresh, but it works.
            perror("Failed to send signal (Try sudo)");
        }
    }
}
